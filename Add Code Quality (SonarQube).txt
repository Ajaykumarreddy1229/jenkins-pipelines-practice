pipeline{
    agent anystagesstages{
        stage("Build"){
            steps{
                sh "mvn clean package"
            }
        }
        stage("test"){
            steps{
                sh mvn test"
            }
        }
        stage("code Qulaity"){
            steps{
                sh "mvn sonar:sonar"
            }
        }
        stage("Deploy"){
            steps{
                sh 'cp target/myapp.war /opt/tomcat/webapps/'
            }
        }
    }
}
