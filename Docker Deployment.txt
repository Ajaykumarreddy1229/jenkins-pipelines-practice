pipeline {
    agent any
    stages {
        stage('Build Docker Image') {
            steps {
                sh 'docker build -t myapp:latest .'
            }
        }
        stage('Push to Registry') {
            steps {
                sh 'docker push myrepo/myapp:latest'
            }
        }
        stage('Deploy with Docker') {
            steps {
                sh 'docker run -d -p 8080:8080 myrepo/myapp:latest'
            }
        }
    }
}
