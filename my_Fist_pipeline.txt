pipeline{
    agent any //run on any jenkins agent
    stages{
        stage{
            steps{
                git 'https://github.com/your-repo/myapp.git'
            }
        }
        stage(Build){
            steps{
                sh 'mvn clean package'
            }
        }
        stage('Test'){
            steps{
                sh 'mvn test'
            }
        }
stage('Deploy'){
    steps{
        sh"""
        cp target/myapp.war /path/to/tomcat/webapps/
        """
    }
}
    }
}
